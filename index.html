let scpDatabase = []; // 1. CARICA I DATI DAL JSON async function loadDatabase() { const status = document.getElementById('search-status'); try { const response = await fetch('database.json'); scpDatabase = await response.json(); renderCards(scpDatabase); status.innerText = `DATABASE ONLINE: ${scpDatabase.length} FILE CARICATI.`; } catch (error) { status.innerText = "ERRORE: IMPOSSIBILE ACCEDERE AL DATABASE."; console.error(error); } } // 2. FUNZIONE PER CREARE LE CARD FISICAMENTE function renderCards(data) { const grid = document.querySelector('.scp-grid'); grid.innerHTML = ""; // Pulisce la griglia data.forEach(scp => { const card = document.createElement('div'); card.className = "scp-entry"; card.innerHTML = ` ${scp.class.toUpperCase()}

${scp.id}

NICKNAME: ${scp.name}

${scp.desc}

ACCESS FILE `; grid.appendChild(card); }); } // 3. FUNZIONE DI RICERCA AGGIORNATA function filterSCP() { let input = document.getElementById('scpSearch').value.toUpperCase(); let filtered = scpDatabase.filter(scp => scp.id.includes(input) || scp.name.toUpperCase().includes(input) ); renderCards(filtered); document.getElementById('search-status').innerText = `TROVATI ${filtered.length} RISULTATI.`; } // Inizializza al caricamento window.onload = loadDatabase;
